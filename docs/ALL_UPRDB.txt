CREATE TABLE dbo.CONTACT
(
ContactID INT IDENTITY(1,1) NOT NULL,
ContactTypeCode        NVARCHAR(30) NOT NULL,
FirstName              NVARCHAR(100) NULL,
LastName               NVARCHAR(100) NULL,
OrganizationName       NVARCHAR(200) NULL,
Email                  NVARCHAR(150) NULL,
Phone                  NVARCHAR(30) NULL,
MailingAddress1        NVARCHAR(200) NULL,
MailingAddress2        NVARCHAR(200) NULL,
City                   NVARCHAR(100)     NULL,
State                  CHAR(2)          NULL,
ZipCode                NVARCHAR(10)   NULL,
IsActive               BIT     NOT NULL CONSTRAINT DF_CONTACT_IsActive
DEFAULT (1),
CreatedDate            DATETIME2(0) NOT NULL CONSTRAINT DF_CONTACT_CreatedDate
DEFAULT (SYSDATETIME()),
UpdatedDate            DATETIME2(0) NOT NULL CONSTRAINT DF_CONTACT_UpdatedDate
DEFAULT (SYSDATETIME()),
/* =========================
PRIMARY KEY
========================= */
CONSTRAINT PK_CONTACT
PRIMARY KEY CLUSTERED (ContactID),
/* =========================
BASIC VALIDATIONS
========================= */
-- Must have either a person name OR organization
CONSTRAINT CK_CONTACT_NameOrOrg
CHECK (
(FirstName IS NOT NULL AND LastName IS NOT NULL)
OR OrganizationName IS NOT NULL
),
-- State must be 2 characters if provided
CONSTRAINT CK_CONTACT_State
CHECK (
State IS NULL OR LEN(State) = 2
),
-- Zip code basic format (5 or 9 digits)
CONSTRAINT CK_CONTACT_Zip
CHECK (
ZipCode IS NULL
OR ZipCode LIKE '[0-9][0-9][0-9][0-9][0-9]'
OR ZipCode LIKE '[0-9][0-9][0-9][0-9][0-9]-%'
),
-- Email basic validation
CONSTRAINT CK_CONTACT_Email
CHECK (
Email IS NULL OR Email LIKE '%@%.%'
),
-- IsActive must be 0 or 1
CONSTRAINT CK_CONTACT_IsActive
CHECK (IsActive IN (0,1)),
-- UpdatedDate must be >= CreatedDate
CONSTRAINT CK_CONTACT_Date
CHECK (UpdatedDate >= CreatedDate)
);
